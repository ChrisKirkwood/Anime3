import logging
import os
import openai
import tiktoken

# Initialize logging
log_file_path = r"D:\Anime3\log\backend.log"
os.makedirs(os.path.dirname(log_file_path), exist_ok=True)

logging.basicConfig(
    level=logging.DEBUG,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler(log_file_path, mode='a'),
        logging.StreamHandler()
    ]
)
logger = logging.getLogger(__name__)

# Initialize tokenizer
tokenizer = tiktoken.get_encoding("cl100k_base")


def clean_subtitles_file(input_file, output_file):
    """
    Simplified pipeline to read subtitles, pass raw data to OpenAI, and save results.
    """
    try:
        # Read raw subtitles
        logger.info(f"Reading subtitles from: {input_file}")
        with open(input_file, 'r', encoding='utf-8') as f:
            raw_subtitles = f.readlines()

        logger.debug(f"Raw content from file:\n{''.join(raw_subtitles)}")

        # Split subtitles into timestamp and content
        subtitles_with_timestamps = []
        current_timestamp = None
        current_subtitle = []

        for idx, line in enumerate(raw_subtitles, 1):
            line = line.strip()
            if ":" in line:  # Likely a timestamp
                if current_timestamp is not None:  # Save the previous subtitle
                    subtitles_with_timestamps.append((current_timestamp, " ".join(current_subtitle)))
                    logger.debug(f"[Line {idx}] Parsed: Timestamp='{current_timestamp}', Subtitle='{' '.join(current_subtitle)}'")
                
                # Start a new subtitle
                try:
                    current_timestamp, subtitle = line.split(":", 1)
                    current_timestamp = current_timestamp.strip()
                    current_subtitle = [subtitle.strip()]
                except ValueError:
                    logger.warning(f"[Line {idx}] Failed to parse line: {line}")
                    continue
            else:  # Continuation of a subtitle
                if current_timestamp is not None:
                    current_subtitle.append(line)
                else:
                    logger.warning(f"[Line {idx}] Unhandled line without timestamp: {line}")

        # Append the last subtitle
        if current_timestamp is not None:
            subtitles_with_timestamps.append((current_timestamp, " ".join(current_subtitle)))

        logger.info(f"Extracted {len(subtitles_with_timestamps)} valid subtitles.")

        # Save raw subtitles to intermediate file
        intermediate_file = r"D:\Anime3\output\pre_openai_subtitles.txt"
        with open(intermediate_file, 'w', encoding='utf-8') as f:
            for timestamp, subtitle in subtitles_with_timestamps:
                f.write(f"{timestamp}: {subtitle}\n")
        logger.info(f"Intermediate file saved: {intermediate_file}")

        # Process with OpenAI
        cleaned_subtitles = process_with_openai(subtitles_with_timestamps)

        # Save cleaned subtitles to output file
        with open(output_file, 'w', encoding='utf-8') as f:
            for subtitle in cleaned_subtitles:
                f.write(subtitle + '\n')
        logger.info(f"Cleaned subtitles saved to: {output_file}")

    except Exception as e:
        logger.error(f"Error in clean_subtitles_file: {e}")


def process_with_openai(subtitles_with_timestamps, max_batch_tokens=12000):
    """
    Processes subtitles in batches, ensuring no batch exceeds token limits.
    """
    try:
        # Batch subtitles
        batches = []
        current_batch = []
        current_tokens = 0

        for timestamp, subtitle in subtitles_with_timestamps:
            subtitle_tokens = len(tokenizer.encode(subtitle))
            timestamp_tokens = len(tokenizer.encode(timestamp))
            total_tokens = subtitle_tokens + timestamp_tokens

            if current_tokens + total_tokens > max_batch_tokens:
                batches.append(current_batch)
                current_batch = []
                current_tokens = 0

            current_batch.append((timestamp, subtitle))
            current_tokens += total_tokens

        if current_batch:
            batches.append(current_batch)

        logger.info(f"Created {len(batches)} batches for OpenAI processing.")

        # Send each batch to OpenAI
        cleaned_subtitles = []
        for i, batch in enumerate(batches, start=1):
            prompt = "\n".join([f"{timestamp}: {subtitle}" for timestamp, subtitle in batch])
            try:
                response = openai.ChatCompletion.create(
                    model="gpt-3.5-turbo",
                    messages=[{"role": "user", "content": prompt}],
                    max_tokens=1500
                )
                cleaned_batch = response['choices'][0]['message']['content'].strip().split("\n")
                cleaned_subtitles.extend(cleaned_batch)
            except Exception as e:
                logger.error(f"Error processing batch {i}: {e}")

        return cleaned_subtitles

    except Exception as e:
        logger.error(f"Error in process_with_openai: {e}")
        return []
