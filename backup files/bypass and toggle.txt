# Normalize and clean text
def normalize_text(text, context="General"):
    """
    Cleans up detected text by removing extraneous characters and normalizing spacing.
    Ensures punctuation is preserved and multi-sentence subtitles are intact.
    
    Args:
        text (str): The input text to normalize.
        context (str): The context of the text (e.g., 'Subtitle', 'Timestamp').
    
    Returns:
        str: The normalized text.
    """
    # Check if the feature is toggled off
    if not FEATURE_TOGGLES.get("normalize_text", True):
        logger.info(f"[{context}] Normalization is disabled. Returning the original text.")
        return text  # Return the original text as is
    
    try:
        logger.debug(f"[{context}] Starting normalization for text: {text}")

        # Record initial state
        initial_length = len(text)
        logger.debug(f"[{context}] Initial text length: {initial_length}")
        
        # Step 1: Normalize spaces
        text_with_normalized_spaces = re.sub(r'\s+', ' ', text)
        logger.debug(f"[{context}] After normalizing spaces: {text_with_normalized_spaces} (Length: {len(text_with_normalized_spaces)})")

        # Step 2: Remove unwanted characters, but retain essential punctuation
        cleaned_text = re.sub(r'[^\w .,!?\'"()\-:;]', '', text_with_normalized_spaces)
        logger.debug(f"[{context}] After removing unwanted characters: {cleaned_text} (Length: {len(cleaned_text)})")

        # Step 3: Trim whitespace
        normalized = cleaned_text.strip()
        logger.debug(f"[{context}] After trimming whitespace: {normalized} (Length: {len(normalized)})")

        # Log summary of changes
        removed_characters_count = initial_length - len(normalized)
        logger.info(f"[{context}] Normalization complete. Original length: {initial_length}, Final length: {len(normalized)}, Characters removed: {removed_characters_count}")

        return normalized
    except Exception as e:
        logger.error(f"[{context}] Error normalizing text: {e}. Input: {text}")
        return text




# Check if text is valid English
def is_valid_english(text):
    """
    Determines if text is valid English using a combination of word and character checks.
    Multi-sentence subtitles are retained based on punctuation and contextual checks.
    Includes enhanced logging to track subtitles through the entire validation process.
    """
    # Check if the feature is toggled off
    if not FEATURE_TOGGLES.get("is_valid_english", True):
        logger.info("English validation is disabled. Assuming the text is valid.")
        return True  # Assume valid if the feature is disabled

    try:
        logger.debug(f"Starting validation for English text: '{text}'")

        original_length = len(text)
        logger.debug(f"Original text length: {original_length}")

        # Debug mode bypass
        if DEBUG_MODE:
            logger.debug(f"DEBUG_MODE ON: Skipping English validation for '{text}'")
            return True

        # Tokenize text and log
        tokens = text.split()
        token_count = len(tokens)
        logger.debug(f"Tokenized text: {tokens} (Token count: {token_count})")

        # Reject empty subtitles and log
        if not tokens:
            logger.info(f"Rejected text: '{text}' - No valid tokens found.")
            return False

        # Word validation
        valid_words = [word for word in tokens if word.lower() in english_vocab or word in whitelist]
        valid_word_count = len(valid_words)
        word_score = valid_word_count / token_count
        logger.debug(f"Valid words: {valid_words} (Valid word count: {valid_word_count}, Word score: {word_score:.2f})")

        # Character validation
        english_chars = set("abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789 .,!?'-")
        char_count = sum(1 for char in text if char in english_chars)
        char_score = char_count / original_length if original_length > 0 else 0
        logger.debug(f"Valid characters: {char_count}/{original_length} (Character score: {char_score:.2f})")

        # Punctuation and multi-sentence check
        punctuation_count = text.count('.') + text.count('!') + text.count('?')
        logger.debug(f"Punctuation count: {punctuation_count} (Periods: {text.count('.')} "
                     f"Exclamations: {text.count('!')} Questions: {text.count('?')})")

        # Retain based on validation thresholds and punctuation
        if word_score > 0.35:
            logger.debug(f"Accepted text: '{text}' - Word score ({word_score:.2f}) exceeds threshold.")
            return True
        elif char_score > 0.65:
            logger.debug(f"Accepted text: '{text}' - Character score ({char_score:.2f}) exceeds threshold.")
            return True
        elif punctuation_count > 0:
            logger.debug(f"Accepted text: '{text}' - Contains valid punctuation.")
            return True
        else:
            logger.info(f"Rejected text: '{text}' - Fails word and character score thresholds and lacks punctuation.")
            return False

    except Exception as e:
        logger.error(f"Error validating English text: {e}. Input text: '{text}'")
        return False




# Simplify subtitle validation
def is_valid_subtitle(subtitle):
    """
    Validates a subtitle's structure and content for cleaning.
    Retains multi-sentence subtitles explicitly, with enhanced logging at each validation step.
    """
    # Check if the feature is toggled off
    if not FEATURE_TOGGLES.get("is_valid_subtitle", True):
        logger.info("Subtitle validation is disabled. Assuming the subtitle is valid.")
        return True  # Assume valid if the feature is disabled

    try:
        logger.debug(f"Starting validation for subtitle: '{subtitle}'")
        original_length = len(subtitle)
        
        # Trim whitespace and log
        subtitle = subtitle.strip()
        logger.debug(f"Trimmed subtitle: '{subtitle}' (Length: {len(subtitle)})")
        
        # Check for empty subtitle
        if not subtitle:
            logger.info(f"Rejected subtitle: '{subtitle}' - Empty after trimming.")
            return False

        # Step 1: Check for minimum length
        if len(subtitle) < 3:
            logger.info(f"Rejected subtitle: '{subtitle}' - Too short (Length: {len(subtitle)})")
            if DEBUG_MODE:
                logger.debug(f"DEBUG_MODE ON: Skipping length check for '{subtitle}'")
                return True
            return False

        # Step 2: Check for known garbage values
        if subtitle.lower() in {"defle", "pororo bory", "sam", "www", "ttt", "ahh", "???", "none"}:
            logger.info(f"Rejected subtitle: '{subtitle}' - Known garbage value.")
            if DEBUG_MODE:
                logger.debug(f"DEBUG_MODE ON: Skipping garbage value check for '{subtitle}'")
                return True
            return False

        # Step 3: Check for lack of alphabetic characters
        if not any(char.isalpha() for char in subtitle):
            logger.info(f"Rejected subtitle: '{subtitle}' - Contains no alphabetic characters.")
            if DEBUG_MODE:
                logger.debug(f"DEBUG_MODE ON: Skipping alphabetic character check for '{subtitle}'")
                return True
            return False

        # Step 4: Check for multi-sentence subtitles
        sentence_count = subtitle.count('.') + subtitle.count('!') + subtitle.count('?')
        logger.debug(f"Subtitle sentence count: {sentence_count} for '{subtitle}'")
        if sentence_count > 1:
            logger.debug(f"Accepted subtitle: '{subtitle}' - Contains multiple sentences (Sentence count: {sentence_count})")
            return True

        # Step 5: Validate English content
        valid = is_valid_english(subtitle)
        if not valid:
            logger.info(f"Rejected subtitle: '{subtitle}' - Fails English validation.")
            if DEBUG_MODE:
                logger.debug(f"DEBUG_MODE ON: Skipping English validation for '{subtitle}'")
                return True
        else:
            logger.debug(f"Accepted subtitle: '{subtitle}' - Passes English validation.")

        # Final log for valid subtitle
        logger.debug(f"Validation complete for subtitle: '{subtitle}' - Accepted (Original Length: {original_length}, Final Length: {len(subtitle)})")
        return valid

    except Exception as e:
        logger.error(f"Error validating subtitle: {e}. Input: '{subtitle}'")
        return False




# Check similarity between subtitles
def is_similar(text1, text2, threshold=0.85):
    """
    Checks similarity between two text strings using Levenshtein ratio.
    Logs the decision-making process for transparency and tracks duplicates.
    """
    # Check if the feature is toggled off
    if not FEATURE_TOGGLES.get("is_similar", True):
        logger.info("Similarity check is disabled. Assuming texts are not similar.")
        return False  # Default to not similar if the feature is disabled

    try:
        logger.debug(f"Starting similarity check between:\n - Text1: '{text1}'\n - Text2: '{text2}'\n"
                     f" - Similarity threshold: {threshold}")

        # Calculate similarity ratio
        similarity = SequenceMatcher(None, text1, text2).ratio()
        logger.debug(f"Computed similarity ratio: {similarity:.2f}")

        # Decision logging based on threshold
        if similarity > threshold:
            logger.info(f"Texts are considered similar:\n"
                        f" - Text1: '{text1}'\n"
                        f" - Text2: '{text2}'\n"
                        f" - Similarity: {similarity:.2f} exceeds threshold ({threshold}).")
            
            # Log duplicate removal explicitly
            logger.info(f"Subtitle '{text2}' is flagged as a duplicate of '{text1}' and will be removed.")
            return True
        else:
            logger.info(f"Texts are NOT similar:\n"
                        f" - Text1: '{text1}'\n"
                        f" - Text2: '{text2}'\n"
                        f" - Similarity: {similarity:.2f} does not meet threshold ({threshold}).")
            return False
    except Exception as e:
        logger.error(f"Error calculating similarity:\n"
                     f" - Text1: '{text1}'\n"
                     f" - Text2: '{text2}'\n"
                     f" - Error: {e}")
        return False

# Function to deduplicate subtitles
def deduplicate_subtitles(subtitles_with_timestamps):
    """
    Deduplicates a list of subtitles by comparing their content for similarity.

    Args:
        subtitles_with_timestamps (list): List of tuples (timestamp, subtitle).

    Returns:
        list: Deduplicated list of tuples (timestamp, subtitle).
    """
    # Check if the feature is toggled off
    if not FEATURE_TOGGLES.get("deduplication", True):
        logger.info("Deduplication is disabled. Returning subtitles as is.")
        return subtitles_with_timestamps  # Return original list if deduplication is disabled

    deduplicated_subtitles = []

    try:
        for idx, current_subtitle in enumerate(subtitles_with_timestamps, 1):
            duplicate_found = False
            for deduplicated_subtitle in deduplicated_subtitles:
                if FEATURE_TOGGLES.get("is_similar", True):  # Check if similarity check is toggled on
                    if is_similar(current_subtitle[1], deduplicated_subtitle[1]):  # Compare subtitles only
                        logger.info(f"[Deduplication] Duplicate detected: '{current_subtitle[1]}' will not be added.")
                        logger.debug(f"[Deduplication] Original: '{deduplicated_subtitle[1]}' | Duplicate: '{current_subtitle[1]}'")
                        duplicate_found = True
                        break
                else:
                    logger.debug("[Deduplication] Similarity check is disabled. Skipping similarity check.")

            if not duplicate_found:
                deduplicated_subtitles.append(current_subtitle)
                logger.debug(f"[Deduplication] Subtitle added: {current_subtitle[1]}")

        logger.info(f"Deduplication complete. Total unique subtitles: {len(deduplicated_subtitles)}")
        return deduplicated_subtitles

    except Exception as e:
        logger.error(f"Error during deduplication: {e}")
        return subtitles_with_timestamps  # Fallback to original list in case of error




# Updated batch_subtitles function
def batch_subtitles(subtitles_with_timestamps, max_batch_tokens=12000):
    """
    Splits subtitles into batches that fit within the token limit.

    Args:
        subtitles_with_timestamps (list): List of subtitles with timestamps.
        max_batch_tokens (int): Maximum number of tokens allowed per batch.

    Returns:
        list: A list of subtitle batches.
    """
    try:
        batches = []
        current_batch = []
        current_tokens = 0

        # Initialize the tokenizer (specific to OpenAI model, e.g., gpt-3.5-turbo)
        tokenizer = tiktoken.get_encoding("cl100k_base")

        logger.info("Starting batch processing...")

        for idx, (timestamp, subtitle) in enumerate(subtitles_with_timestamps, start=1):
            # Estimate token count for subtitle and timestamp
            subtitle_tokens = len(tokenizer.encode(subtitle))
            timestamp_tokens = len(tokenizer.encode(timestamp))
            total_tokens = subtitle_tokens + timestamp_tokens

            # Log token counts
            logger.debug(f"[Subtitle {idx}] Timestamp Tokens: {timestamp_tokens}, Subtitle Tokens: {subtitle_tokens}, "
                         f"Total Tokens: {total_tokens} for '{timestamp}: {subtitle}'")

            # Check if adding this subtitle exceeds the batch token limit
            if current_tokens + total_tokens > max_batch_tokens:
                if not current_batch:
                    logger.error(f"Single subtitle exceeds max_batch_tokens: {total_tokens}. Skipping subtitle.")
                else:
                    batches.append(current_batch)
                    logger.debug(f"Created batch with {len(current_batch)} subtitles.")
                    current_batch = []
                    current_tokens = 0

            # Add subtitle to the current batch
            current_batch.append((timestamp, subtitle))
            current_tokens += total_tokens
            logger.debug(f"[Subtitle {idx}] Added to batch. Current batch size: {len(current_batch)}, "
                         f"Current Tokens: {current_tokens}.")

        # Add any remaining subtitles to the final batch
        if current_batch:
            logger.debug(f"Final batch created with {len(current_batch)} subtitles. Total Tokens: {current_tokens}.")
            batches.append(current_batch)

        logger.info(f"Batch processing completed. Total batches created: {len(batches)}.")

        # Log batch contents
        for batch_idx, batch in enumerate(batches, start=1):
            logger.debug(f"[Batch {batch_idx}] contains {len(batch)} subtitles:")
            for subtitle_idx, (timestamp, subtitle) in enumerate(batch, start=1):
                logger.debug(f"    [Batch {batch_idx}, Subtitle {subtitle_idx}] {timestamp}: {subtitle}")

        return batches

    except Exception as e:
        logger.error(f"Error during batch processing: {e}")
        return []





# Use OpenAI API to clean a batch of subtitles
def clean_subtitles_with_openai_batch(subtitles_with_timestamps, max_batch_tokens=12000):
    """
    Cleans subtitles in batches, adhering to token limits, and retains your custom prompt logic.

    Args:
        subtitles_with_timestamps (list): List of subtitles with timestamps.
        max_batch_tokens (int): Maximum number of tokens allowed per batch.

    Returns:
        list: A list of cleaned subtitles.
    """
    # Check if the feature is toggled off
    if not FEATURE_TOGGLES.get("clean_with_openai", True):
        logger.info("OpenAI cleaning is disabled. Returning subtitles as is.")
        return [f"{timestamp}: {subtitle}" for timestamp, subtitle in subtitles_with_timestamps]

    try:
        # Function to split subtitles into batches
        def batch_subtitles(subtitles, max_tokens):
            """
            Helper function to batch subtitles into manageable groups within token limits.
            """
            batches = []
            current_batch = []
            current_tokens = 0

            for timestamp, subtitle in subtitles:
                subtitle_tokens = len(subtitle.split())
                timestamp_tokens = len(timestamp.split())
                total_tokens = subtitle_tokens + timestamp_tokens

                if current_tokens + total_tokens > max_tokens:
                    logger.debug(f"Batch created with {len(current_batch)} subtitles (tokens: {current_tokens}).")
                    batches.append(current_batch)
                    current_batch = []
                    current_tokens = 0

                current_batch.append((timestamp, subtitle))
                current_tokens += total_tokens

            if current_batch:
                logger.debug(f"Final batch created with {len(current_batch)} subtitles (tokens: {current_tokens}).")
                batches.append(current_batch)

            return batches

        # Split the subtitles into batches
        logger.info("Starting batch creation...")
        batches = batch_subtitles(subtitles_with_timestamps, max_batch_tokens)
        logger.info(f"{len(batches)} batches created for cleaning.")
        logger.debug(f"Batches content:\n{batches}")

        cleaned_subtitles = []

        for i, batch in enumerate(batches):
            logger.info(f"Processing batch {i+1}/{len(batches)} with {len(batch)} subtitles.")

            # Log the batch being sent
            logger.debug(f"Batch {i+1} contents:")
            for timestamp, subtitle in batch:
                logger.debug(f"[Batch {i+1}] Subtitle: [{timestamp}] {subtitle}")

            # Construct the batch-specific prompt
            prompt = (
                "You are a subtitle cleaning assistant. For each subtitle below, follow these instructions:\n\n"
                "### Content Preservation:\n"
                "1. Retain all non-English words or phrases (e.g., names, places, culturally significant terms) if they are meaningful and necessary for the subtitle.\n"
                "2. Preserve all sentences within a subtitle block without removing meaningful content or context.\n"
                "3. Do not truncate subtitles unless they are nonsensical or excessively verbose.\n"
                "4. Ensure all meaningful text is preserved, even if repeated within close proximity.\n\n"
                "### Formatting and Grammar:\n"
                "5. Remove unnecessary special characters or formatting artifacts, but retain punctuation necessary for proper grammar and meaning.\n"
                "6. Standardize capitalization where appropriate (e.g., names, the start of sentences), but retain intentional stylistic choices.\n"
                "7. Remove typographical errors or artifacts, such as 'PORORO BORY'.\n\n"
                "### Deduplication and Timestamps:\n"
                "8. Identify and merge duplicates (identical or nearly identical subtitles) into a single entry with one timestamp.\n"
                "9. Use the earliest timestamp associated with duplicates and discard others.\n"
                "10. Do not create new timestamps or introduce timestamp ranges (e.g., 'start-end').\n\n"
                "### Style and Consistency:\n"
                "11. Maintain a consistent tone throughout all subtitles. Preserve the original style, whether formal or casual.\n\n"
                "### Example:\n"
                "Input:\n"
                "0:01: PORORO BORY!!!\n"
                "0:02: PORORO BORY!!!\n"
                "0:03: It's the break of dawn.\n\n"
                "Output:\n"
                "0:01: It's the break of dawn. [Merged duplicate lines into meaningful context]\n\n"
            )

            # Add the subtitles from the current batch
            for timestamp, subtitle in batch:
                prompt += f"{timestamp}: {subtitle}\n"

            logger.debug(f"Constructed prompt for batch {i+1}:\n{prompt}")

            try:
                response = openai.ChatCompletion.create(
                    model="gpt-3.5-turbo",
                    messages=[
                        {"role": "system", "content": "You are a helpful assistant."},
                        {"role": "user", "content": prompt}
                    ],
                    temperature=0.0,
                    max_tokens=1500
                )

                # Log raw OpenAI response
                cleaned_text = response['choices'][0]['message']['content'].strip()
                logger.debug(f"OpenAI raw response for batch {i+1}:\n{cleaned_text}")

                # Extract cleaned subtitles
                for line in cleaned_text.split("\n"):
                    if ":" in line:
                        cleaned_subtitles.append(line.strip())
                        logger.debug(f"Cleaned Subtitle: {line.strip()}")

            except Exception as e:
                logger.error(f"Error cleaning batch {i+1}: {e}")
                continue

        # Log final cleaned subtitles
        logger.info(f"Finished cleaning all batches. Total cleaned subtitles: {len(cleaned_subtitles)}")
        logger.debug(f"Cleaned subtitles:\n{cleaned_subtitles}")
        return cleaned_subtitles

    except Exception as e:
        logger.error(f"Error cleaning subtitles in batches: {e}")
        return []
